```php
namespace App\Http\Controllers\Api;

        return response()->json($form);

    public function index()

        $data = $request->validate([
            'title' => 'required',

            'is_public' => 'boolean',
            'is_active' => 'boolean',
            'template_id' => 'nullable|exists:form_templates,id',
        ]);

        $form = Form::create($data);

        return response()->json($form, 201);
    }

    // ADMIN: Update form
    public function update(Request $request, Form $form)
    {
        $data = $request->validate([
            'title' => 'sometimes',
            'slug' => 'sometimes|unique:forms,slug,' . $form->id,
            'description' => 'nullable',
            'institution_id' => 'nullable|exists:institutions,id',
            'is_public' => 'boolean',
            'is_active' => 'boolean',
            'template_id' => 'nullable|exists:form_templates,id',
        ]);

        $form->update($data);

        return response()->json($form);
    }

    // ADMIN: Delete form
    public function destroy(Form $form)
    {
        $form->delete();
```

```php
<?php

namespace App\Http\Controllers\Api;

use App\Models\Form;
use App\Models\FormSubmission;
use Illuminate\Http\Request;
use App\Http\Controllers\Controller;

class FormSubmissionController extends Controller
{
    // PUBLIC: Submit form
    public function store(Request $request, $slug)
    {
        $form = Form::where('slug', $slug)->where('is_active', true)->firstOrFail();

        $data = $request->validate([
            'fields' => 'required|array',
            'files.*' => 'file|max:10240',
        ]);

        $submission = FormSubmission::create([
            'form_id' => $form->id,
            'submitted_by' => auth()->id(),
            'institution_id' => $form->institution_id,
            'data' => $data['fields'],
            'payment_status' => $form->payment_enabled ? 'pending' : 'paid',
            'ip_address' => $request->ip(),
            'user_agent' => $request->userAgent(),
        ]);

        // Handle file uploads
        if ($request->hasFile('files')) {
            foreach ($request->file('files') as $fieldId => $file) {
                $path = $file->store('form_uploads');

                $submission->files()->create([
                    'field_id' => $fieldId,
                    'file_path' => $path,
                ]);
            }
        }

        return response()->json([
            'message' => 'Form submitted successfully',
            'submission_id' => $submission->id,
        ]);
    }

    // ADMIN: List submissions
    public function index(Form $form)
    {
        return $form->submissions()->paginate(20);
    }

    // ADMIN: View submission
    public function show(FormSubmission $submission)
    {
        return response()->json($submission->load('files'));
    }
}
```

```php
use App\Models\FormTemplate;
use Illuminate\Http\Request;
use App\Http\Controllers\Controller;

class FormTemplateController extends Controller
{
    public function index()
    {
        return FormTemplate::all();
    }

    public function store(Request $request)
    {
        $data = $request->validate([
            'name' => 'required',
            'category' => 'nullable|string',
            'description' => 'nullable|string',
            'fields' => 'array',
            'ai_generated' => 'boolean',
        ]);

        $template = FormTemplate::create($data);

        return response()->json($template, 201);
    }

    public function update(Request $request, FormTemplate $template)
    {
        $template->update($request->all());
        return response()->json($template);
    }

    public function destroy(FormTemplate $template)
    {
        $template->delete();
        return response()->json(['message' => 'Template deleted']);
    }
}
```

# ⭐ 10. Alpine.js Form Builder Logic

Add this to the bottom of `builder.blade.php`:

```blade
<script>
function formBuilder(formId) {
    return {
        fields: [],
        editingField: null,
        aiPrompt: '',
        aiResult: null,

        init() {
            axios.get(`/api/admin/forms/${formId}/fields`)
                .then(res => this.fields = res.data);
        },

        addField() {
            axios.post(`/api/admin/forms/${formId}/fields`, {
                type: 'text',
                label: 'New Field'
            }).then(res => this.fields.push(res.data));
        },

        editField(field) {
            this.editingField = JSON.parse(JSON.stringify(field));
        },

        saveField() {
            axios.put(`/api/admin/fields/${this.editingField.id}`, this.editingField)
                .then(() => {
                    let index = this.fields.findIndex(f => f.id === this.editingField.id);
                    this.fields[index] = this.editingField;
                    this.editingField = null;
                });
        },

        deleteField(id) {
            axios.delete(`/api/admin/fields/${id}`)
                .then(() => this.fields = this.fields.filter(f => f.id !== id));
        },

        addLogicRule() {
            if (!this.editingField.conditional_logic) {
                this.editingField.conditional_logic = { rules: [] };
            }
            this.editingField.conditional_logic.rules.push({
                field_id: null,
                operator: 'equals',
                value: ''
            });
        },

        removeLogicRule(index) {
            this.editingField.conditional_logic.rules.splice(index, 1);
        },

        generateWithAI() {
            axios.post('/mcp/run', {
                tool: 'add_field_to_form',
                prompt: this.aiPrompt,
                form_id: formId
            }).then(res => {
                this.aiResult = res.data;
            });
        }
    }
}
</script>
```

Add this to the bottom of `show.blade.php`:

```blade
<script>
function publicForm(formId, slug) {
    return {
        fields: [],
        formData: {},
        files: {},
        loading: false,

        init() {
            axios.get(`/api/forms/${slug}`)
                .then(res => {
                    this.fields = res.data.fields;
                    this.fields.forEach(f => this.formData[f.id] = '');
                });
        },

        shouldShowField(field) {
            if (!field.conditional_logic) return true;

            return field.conditional_logic.rules.every(rule => {
                let val = this.formData[rule.field_id];
                if (rule.operator === 'equals') return val == rule.value;
                if (rule.operator === 'not_equals') return val != rule.value;
                if (rule.operator === 'contains') return val?.includes(rule.value);
                return true;
            });
        },

        handleFileUpload(event, fieldId) {
            this.files[fieldId] = event.target.files[0];
        },

        submitForm() {
            this.loading = true;

            let form = new FormData();
            form.append('fields', JSON.stringify(this.formData));

            Object.keys(this.files).forEach(id => {
                form.append(`files[${id}]`, this.files[id]);
            });

            axios.post(`/forms/${slug}`, form)
                .then(() => window.location = `/forms/${slug}/success`)
                .finally(() => this.loading = false);
        }
    }
}
</script>
```

# 3. **Controller Logic for Multi‑Page Navigation**

`app/Http/Controllers/PublicFormController.php`

```php
public function page($slug, $page)
{
    $form = Form::with(['fields', 'fields.group', 'groups'])->where('slug', $slug)->firstOrFail();

    $groups = $form->groups()->orderBy('order')->get();

    $currentGroup = $groups[$page - 1] ?? abort(404);

    $fields = $form->fields()->where('group_id', $currentGroup->id)->orderBy('order')->get();

    return view('public.forms.page', [
        'form' => $form,
        'groups' => $groups,
        'currentGroup' => $currentGroup,
        'fields' => $fields,
        'page' => $page,
        'totalPages' => $groups->count(),
    ]);
}

public function pageSubmit(Request $request, $slug, $page)
{
    $form = Form::where('slug', $slug)->firstOrFail();

    // Store partial data in session
    $sessionKey = "form_{$form->id}_data";
    $existing = session($sessionKey, []);
    $incoming = $request->input('fields', []);

    session([$sessionKey => array_merge($existing, $incoming)]);

    // Determine next page
    $nextPage = $page + 1;

    if ($nextPage > $form->groups()->count()) {
        // Final submission
        $request->merge(['fields' => session($sessionKey)]);
        return $this->submit($request, $slug);
    }

    return redirect()->route('forms.page', [$slug, $nextPage]);
}
```

# ✅ 2. Payment Controller

`app/Http/Controllers/PublicFormPaymentController.php`

```php
<?php

namespace App\Http\Controllers;

use App\Models\Form;
use App\Models\FormSubmission;
use Illuminate\Http\Request;
use Stripe\Stripe;
use Stripe\Checkout\Session as StripeSession;

class PublicFormPaymentController extends Controller
{
    public function show($slug)
    {
        $form = Form::where('slug', $slug)->firstOrFail();

        return view('public.forms.payment', compact('form'));
    }

    public function createCheckout(Request $request, $slug)
    {
        $form = Form::where('slug', $slug)->firstOrFail();

        $sessionKey = "form_{$form->id}_data";
        $submissionData = session($sessionKey);

        // Create submission in "pending" state
        $submission = FormSubmission::create([
            'form_id' => $form->id,
            'data' => $submissionData,
            'payment_status' => 'pending',
            'ip_address' => $request->ip(),
            'user_agent' => $request->userAgent(),
        ]);

        Stripe::setApiKey(config('services.stripe.secret'));

        $checkout = StripeSession::create([
            'payment_method_types' => ['card'],
            'line_items' => [[
                'price_data' => [
                    'currency' => $form->payment_currency ?? 'usd',
                    'product_data' => [
                        'name' => $form->title,
                    ],
                    'unit_amount' => intval($form->payment_amount * 100),
                ],
                'quantity' => 1,
            ]],
            'mode' => 'payment',
            'success_url' => route('forms.payment.success', $slug) . '?session_id={CHECKOUT_SESSION_ID}',
            'cancel_url' => route('forms.payment.cancel', $slug),
            'metadata' => [
                'submission_id' => $submission->id,
            ],
        ]);

        return redirect($checkout->url);
    }

    public function success(Request $request, $slug)
    {
        $sessionId = $request->query('session_id');

        Stripe::setApiKey(config('services.stripe.secret'));
        $session = StripeSession::retrieve($sessionId);

        $submissionId = $session->metadata->submission_id;

        $submission = FormSubmission::findOrFail($submissionId);
        $submission->update([
            'payment_status' => 'paid',
            'payment_transaction_id' => $sessionId,
        ]);

        return redirect()->route('forms.success', $slug);
    }

    public function cancel($slug)
    {
        return redirect()->route('forms.page', [$slug, 1])
            ->with('error', 'Payment was cancelled.');
    }
}
```

# **3. OCR Service (Tesseract + Ghostscript)**

Create:


```php
<?php

namespace App\Services;

use thiagoalessio\TesseractOCR\TesseractOCR;

class OcrService
{
    public function extractText(string $path): string
    {
        // If PDF → convert to TIFF or PNG first
        if (str_ends_with(strtolower($path), '.pdf')) {
            $imagePath = $this->convertPdfToImage($path);
        } else {
            $imagePath = $path;
        }

        return (new TesseractOCR($imagePath))
            ->lang('eng')
            ->psm(3)
            ->oem(1)
            ->run();
    }

    private function convertPdfToImage(string $pdfPath): string
    {
        $output = storage_path('app/tmp/' . uniqid() . '.png');

        // Ghostscript conversion
        $cmd = "gs -dNOPAUSE -dBATCH -sDEVICE=png16m -r300 -sOutputFile={$output} {$pdfPath}";
        shell_exec($cmd);

        return $output;
    }
}
```

# ⭐ 5. Laravel Implementation

`app/Services/CrmSyncService.php`

```php
<?php

namespace App\Services;

use App\Models\Contact;
use App\Models\FormSubmission;
use App\Models\SacramentRecord;

class CrmSyncService
{
    public function sync(FormSubmission $submission, array $data)
    {
        $contact = $this->syncContact($submission, $data);
        $this->syncInstitution($submission, $data, $contact);
        $this->syncSacraments($submission, $data, $contact);
        $this->syncSafeEnvironment($submission, $data, $contact);
        $this->syncCaseManagement($submission, $data, $contact);

        return $contact;
    }

    private function syncContact(FormSubmission $submission, array $data)
    {
        $fields = $data['fields'] ?? [];

        // Identify or create contact
        $contact = app(ContactMatcher::class)->matchOrCreate($fields);

        // Update contact fields
        app(ContactUpdater::class)->update($contact, $fields, $data['confidence'] ?? 'medium');

        // Link submission
        $submission->update(['contact_id' => $contact->id]);

        return $contact;
    }

    private function syncInstitution(FormSubmission $submission, array $data, Contact $contact)
    {
        if (!isset($data['fields']['institution_name'])) return;

        $institution = app(InstitutionMatcher::class)->match($data['fields']);

        if ($institution) {
            $submission->update(['institution_id' => $institution->id]);
            $contact->institutions()->syncWithoutDetaching([$institution->id]);
        }
    }

    private function syncSacraments(FormSubmission $submission, array $data, Contact $contact)
    {
        if (!isset($data['fields']['sacrament'])) return;

        app(SacramentSyncService::class)->sync($contact, $data['fields'], $submission);
    }

    private function syncSafeEnvironment(FormSubmission $submission, array $data, Contact $contact)
    {
```

```php
                ->first();
            if ($contact) return $contact;
        }

        // 4. Fallback: create new
        return Contact::create([
            'full_name'      => $name,
            'email'          => $email,
            'phone'          => $phone,
            'date_of_birth'  => $dob,
            'source'         => 'form_submission',
```

## **2.2. Alpine.js diff generation**

Use a lightweight JS diff library (diff.js or diff-match-patch).  
Here’s a simple inline diff using diff-match-patch:

```blade
<script src="https://unpkg.com/diff-match-patch"></script>

<script>
function reviewQueue() {
    return {
        item: null,
        diffHtml: '',

        openDrawer(id) {
            axios.get(`/admin/review/${id}`)
                .then(res => {
                    this.item = res.data;
                    this.generateDiff();
                    new bootstrap.Offcanvas(this.$refs.drawer).show();
                });
        },

        generateDiff() {
            const dmp = new diff_match_patch();
            const diffs = dmp.diff_main(this.item.old_value || '', this.item.new_value || '');
            dmp.diff_cleanupSemantic(diffs);
            this.diffHtml = dmp.diff_prettyHtml(diffs);
        }
    }
}
</script>
```

This produces a beautiful diff:

- **Green** = additions  
- **Red** = deletions  
- **Grey** = unchanged  

Perfect for reviewing long text fields.

# 🌟 3. Notification Router Service

`app/Services/NotificationRouter.php`

```php
class NotificationRouter
{
    public function send(User $user, $notification)
    {
        $settings = $user->notificationSettings;

        $channels = [];

        if ($settings->email)      $channels[] = 'mail';
        if ($settings->sms)        $channels[] = 'twilio_sms';
        if ($settings->telegram)   $channels[] = 'telegram';
        if ($settings->browser)    $channels[] = 'browser';
        if ($settings->phone_call) $channels[] = 'twilio_call';

        $user->notify((new $notification)->via($channels));
    }
}
```

```php
<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;

class NotificationSettingsController extends Controller
{
    public function edit(Request $request)
    {
        $user = $request->user();
        $settings = $user->notificationSettings; // via accessor

        return view('admin.notifications.settings', compact('user', 'settings'));
    }

    public function update(Request $request)
    {
        $user = $request->user();
        $settings = $user->notificationSettings;

        $data = $request->validate([
            'email'           => 'nullable|boolean',
            'sms'             => 'nullable|boolean',
            'telegram'        => 'nullable|boolean',
            'browser'         => 'nullable|boolean',
            'phone_call'      => 'nullable|boolean',
            'telegram_chat_id'=> 'nullable|string|max:255',
            'phone_number'    => 'nullable|string|max:50',
        ]);

        // checkboxes: if not present, they’re false
        foreach (['email','sms','telegram','browser','phone_call'] as $flag) {
            $data[$flag] = $request->has($flag);
        }

        $settings->update($data);

        return redirect()
            ->route('admin.notifications.settings.edit')
            ->with('success', 'Notification settings updated.');
    }
}
```

# ⭐ 3. Updated Notification Router (Institution‑Aware)

```php
class NotificationRouter
{
    public function sendForInstitution(
        Institution $institution,
        string $eventKey,
        Notification $notification
    ) {
        $event = NotificationEvent::where('key', $eventKey)->first();
        if (!$event) return;

        // 1. Find routing rules for this institution + event
        $rule = InstitutionNotificationRule::where('institution_id', $institution->id)
            ->where('event_id', $event->id)
            ->first();

        if (!$rule) return;

        // 2. Determine recipients
        $users = User::whereIn('id', $rule->user_ids)->get();

        foreach ($users as $user) {
            $channels = $this->resolveChannels($user, $rule);
            if (!empty($channels)) {
                $notification->setChannels($channels);
                $user->notify($notification);
            }
        }

        // 3. Schedule escalation if configured
        if ($rule->escalate_after_minutes && $rule->escalate_to_user_id) {
            dispatch(new EscalateNotificationJob(
                $rule->escalate_to_user_id,
                $eventKey,
                $notification
            ))->delay(now()->addMinutes($rule->escalate_after_minutes));
        }
    }

    private function resolveChannels(User $user, InstitutionNotificationRule $rule)
    {
        // If rule overrides channels, use those
        if (!is_null($rule->email)) {
            return array_filter([
                $rule->email      ? 'mail'        : null,
                $rule->sms        ? 'twilio_sms'  : null,
                $rule->telegram   ? 'telegram'    : null,
                $rule->browser    ? 'browser'     : null,
                $rule->phone_call ? 'twilio_call' : null,
            ]);
        }

        // Otherwise fall back to user matrix
        $matrix = UserNotificationMatrix::where('user_id', $user->id)
            ->where('event_id', $rule->event_id)
            ->first();

        if (!$matrix) return [];

        return array_filter([
            $matrix->email      ? 'mail'        : null,
            $matrix->sms        ? 'twilio_sms'  : null,
            $matrix->telegram   ? 'telegram'    : null,
            $matrix->browser    ? 'browser'     : null,
            $matrix->phone_call ? 'twilio_call' : null,
        ]);
    }
}
```

# ⭐ 1. The Escalation Job  
This job runs after a delay and checks whether the original notification was acted upon.

Create:

```
app/Jobs/EscalateNotificationJob.php
```

```php
<?php

namespace App\Jobs;

use App\Models\User;
use App\Models\NotificationEvent;
use App\Models\InstitutionNotificationRule;
use App\Services\NotificationRouter;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;

class EscalateNotificationJob implements ShouldQueue
{
    use Queueable;

    public function __construct(
        public int $escalateToUserId,
        public string $eventKey,
        public array $payload, // context for the notification
        public int $ruleId
    ) {}

    public function handle()
    {
        $user = User::find($this->escalateToUserId);
        if (!$user) return;

        $rule = InstitutionNotificationRule::find($this->ruleId);
        if (!$rule) return;

        // Check if the item was already resolved
        if ($this->isResolved($this->payload)) {
            return;
        }

        // Send escalation
        app(NotificationRouter::class)->send(
            $user,
            $this->eventKey,
            new \App\Notifications\EscalatedNotification($this->payload)
        );
    }

    private function isResolved(array $payload): bool
    {
        // Example: Review Queue item
        if (isset($payload['review_item_id'])) {
            $item = \App\Models\CrmReviewItem::find($payload['review_item_id']);
            return $item && $item->status !== 'pending';
        }

        // Example: Case intake
        if (isset($payload['case_id'])) {
            $case = \App\Models\CaseRecord::find($payload['case_id']);
            return $case && $case->status !== 'open';
        }

        // Example: Safe Environment
        if (isset($payload['safe_env_id'])) {
            $record = \App\Models\SafeEnvironmentRecord::find($payload['safe_env_id']);
            return $record && $record->is_verified;
        }

        return false;
    }
}
```

This job:

- Runs after a delay  
- Checks if the item is still unresolved  
- If unresolved → escalates  
- If resolved → silently exits  

This prevents double‑notifications and avoids noise.

```php
Schema::create('notification_logs', function (Blueprint $table) {
    $table->id();

    $table->foreignId('user_id')->nullable()->constrained()->nullOnDelete();
    $table->foreignId('institution_id')->nullable()->constrained()->nullOnDelete();
    $table->foreignId('event_id')->nullable()->constrained('notification_events')->nullOnDelete();

    $table->string('channel'); // mail, sms, telegram, browser, phone_call
    $table->string('status')->default('sent'); // sent, delivered, failed
    $table->string('provider_message_id')->nullable(); // Twilio SID, SendGrid ID, etc.

    $table->json('payload')->nullable(); // context for the notification
    $table->timestamp('sent_at')->nullable();
    $table->timestamp('delivered_at')->nullable();
    $table->timestamp('failed_at')->nullable();

    $table->timestamps();
});
```

This gives you a **complete audit trail**.

```php
class NotificationAnalyticsController extends Controller
{
    public function index()
    {
        $logs = NotificationLog::query();

        // KPI metrics
        $total = $logs->count();
        $delivered = $logs->where('status', 'delivered')->count();
        $failed = $logs->where('status', 'failed')->count();

        // Channel distribution
        $byChannel = NotificationLog::selectRaw('channel, COUNT(*) as count')
            ->groupBy('channel')
            ->pluck('count', 'channel');

        // Event distribution
        $byEvent = NotificationLog::selectRaw('event_id, COUNT(*) as count')
            ->groupBy('event_id')
            ->pluck('count', 'event_id');

        // Institution distribution
        $byInstitution = NotificationLog::selectRaw('institution_id, COUNT(*) as count')
            ->groupBy('institution_id')
            ->pluck('count', 'institution_id');

        // Delivery timeline (last 30 days)
        $timeline = NotificationLog::selectRaw('DATE(sent_at) as date, COUNT(*) as count')
            ->where('sent_at', '>=', now()->subDays(30))
            ->groupBy('date')
            ->orderBy('date')
            ->get();

        return view('admin.notifications.analytics', compact(
            'total', 'delivered', 'failed',
            'byChannel', 'byEvent', 'byInstitution', 'timeline'
        ));
    }
}
```

**1. Public Velzon layer (`/register`, `/donate`, `/forms`, `/profile`)**

- **Use Velzon** for:
  - **eCommerce:** registrations, donations, fees, tickets.
  - **Advanced forms:** multi-step, conditional logic, file uploads.
  - **Simple, pastoral UI:** minimal fields, clear language, mobile-friendly.
- **Output:** clean Blade views using Velzon components, posting into Laravel controllers/services.

---

**2. Application admin Velzon layer (`/admin/...`)**

- **Use Velzon** for all *workflow* and *insight* UIs:
  - **AI Review Queue** (table + drawer + diff view + batch review).
  - **Notification Preferences** (per-user channels).
  - **Per-event Notification Matrix** (user × event × channel).
  - **Institution Routing** (institution × event → recipients + escalation).
  - **Escalation Dashboard** (pending escalations, SLA breaches).
  - **Notification Log Dashboard** (table + filters).
  - **Notification Analytics** (charts, KPIs, timelines).
  - **Email UI** (campaigns, templates, logs).
  - **eCommerce admin** (orders, payments, refunds, reports).

Antigravity rule:  
> If it’s a *workflow*, *dashboard*, *matrix*, *analytics*, *routing*, *AI review*, or *public-ish admin*, build it in Velzon.

- **Services:**
  - `OcrPipelineService`

  - `CrmSyncService`

Antigravity rule:
